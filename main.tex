\begin{frame}{El gasto se ajusta en contratación y personal}
\begin{itemize}\bulletspace
\item \textbf{OPS.} El gasto se reduce 50\% en 2027 frente a 2026, en las entidades del presupuesto nacional. Desde 2028 puede crecer hasta la inflación.
\item \textbf{Personal.} El gasto nominal se reduce 20\% en 2028 frente a 2027, excluyendo a la Fuerza Pública. En 2029 y 2030, el límite es la meta de inflación más 0,5 puntos.
\end{itemize}
\take{El ahorro neto depende de cómo se ejecuten los recortes.}
\src{\ley{227}{Artículos 16 y 17}. Suprimir cargos o contratos puede implicar costos de transición y afectar servicios. El ahorro no equivale automáticamente al porcentaje de la apropiación.}
\end{frame}

\begin{frame}{Renta incorpora una tarifa marginal del 9\%}
\begin{tabular}{lrr}
\toprule
\textbf{Base gravable anual} & \textbf{Hoy} & \textbf{Proyecto}\\
\midrule
Hasta 720 UVT & 0\% & 0\%\\
Más de 720 y hasta 1.090 UVT & 0\% & 9\%\\
\bottomrule
\end{tabular}
\par\medskip
720 UVT = \$\PisoAnual{} al año, o \$\PisoMensual{} por mes.\\
1.090 UVT = \$\TechoAnual{} al año, o \$\TechoMensual{} por mes.
\take{Son ingresos gravables después de la depuración. No son salarios brutos.}
\small Ejemplo con base de 900 UVT: impuesto de $(900-720)\times9\%=16,2$ UVT. Equivale a \$\EjemploImpuesto{} al año o \$\EjemploMes{} por mes.
\src{\ley{258}{Artículo 48}. Conversión con UVT 2026. El equivalente mensual divide la base anual entre 12. La tarifa del tramo solo se aplica al exceso sobre 720 UVT.}
\end{frame}

\begin{frame}{SENA e ICBF dependerían más de la nómina}
\begin{itemize}\bulletspace
\item Las mayores contribuciones empresariales sustituyen parte de los recursos que hoy aporta la Nación.
\item El artículo 106 deroga el artículo 243 del ET, que reserva una parte de renta empresarial y establece garantías de financiación.
\item El aporte adicional \textbf{no implica un aumento equivalente del presupuesto total} de estas entidades.
\end{itemize}
\take{El proyecto aumenta sus recursos propios y flexibiliza la transferencia de la Nación.}
\src{\ley{50}{Exposición, pp. 50--51}, y \ley{333}{artículo 106}. \href{https://normograma.sena.edu.co/compilacion/docs/estatuto_tributario_pr010.htm}{ET 243} asigna 2,2 puntos de tarifa al ICBF y 1,4 al SENA. \begin{frame}{La reforma amplía el IVA y devuelve una parte}
\begin{itemize}\bulletspace
\item Desde 2028 se eliminan las tarifas reducidas de 5\% de los artículos 468-1 y 468-3 y se revisan los bienes excluidos.
\item Se conservan tratamientos especiales para una selección de bienes y servicios esenciales.
\item Los hogares sin devolución asumen el mayor IVA sobre las compras afectadas, según su traslado a precios.
\end{itemize}
\take{Un bien con precio sin IVA de \$100.000 pasa de \$105.000 a \$119.000 si cambia de 5\% a 19\%.}
\src{\ley{248}{Artículos 42 y siguientes}, y \ley{333}{artículo 106}. Ejemplo con precio sin IVA constante: aumento del precio final de 13,3\%. Algunas modificaciones de IVA rigen antes de 2028. No todos los bienes pasan al 19\%.}
\end{frame}

\begin{frame}{EfectIVA depende del IVA atribuible a cada persona}
\begin{itemize}\bulletspace
\item Desde 2028 devuelve 100\% a pobres y al tramo de menores ingresos de población vulnerable. Otros tramos reciben 50\%, 40\% o 30\%.
\item El techo legal mensual es de hasta 5,5 UVT por beneficiario, equivalente a \textbf{\$\DevolucionTope{}} con UVT 2026.
\item La pobreza extrema tiene un monto mínimo, aun sin IVA registrado. La reglamentación fija su valor.
\end{itemize}
\take{El reto operativo es identificar las compras y pagar la devolución completa y a tiempo.}
\src{\ley{241}{Artículos 38--40}. Tope y tramos sujetos a reglas de actualización y reglamentación. Se incluyen facturas y documentos equivalentes. No se exige tarjeta de manera general para compras al contado.}
\end{frame}

\begin{frame}{El IVA llega a más envíos de bajo valor}
\begin{itemize}\bulletspace
\item Hoy ciertos envíos de hasta \textbf{US\$200} desde EE.UU. ingresan sin IVA, según las condiciones del tratado.
\item Según Inexmoda, parte de los pedidos de \textbf{Shein y Temu sale de EE.UU.}, aunque los bienes se fabriquen en China.
\item El proyecto elimina esa exclusión. Para un bien gravado al 19\%, una base de US\$100 genera US\$19 de IVA.
\end{itemize}
\take{DIAN proyecta \$1,7 billones en 2027 para el conjunto de importaciones afectadas, no solo estas plataformas.}
\src{\ley{333}{Art. 106.4.a}, desde la vigencia general. \ley{96}{Exposición, pp. 96--98}, estimación pendiente de reproducción. Ruta logística reportada por \href{https://inexmoda.org.co/stories/temu-y-shein-como-cambia-el-comercio-global-en-2026/}{Inexmoda (2026)}. El arancel tiene reglas distintas. La aplicación del IVA debe conciliarse con el TLC.}
\end{frame}

\begin{frame}{La normalización ofrece un trato especial al incumplimiento}
\begin{itemize}\bulletspace
\item Quien tenga activos omitidos o pasivos inexistentes a enero de 2027 puede normalizarlos con una tarifa de \textbf{15\%}.
\item La inclusión no genera renta por el activo omitido y excluye sanciones y acción penal por la omisión normalizada.
\item Un activo omitido con base de \textbf{\$1.000 millones} genera \textbf{\$150 millones} de normalización.
\end{itemize}
\take{Recauda una vez y plantea un problema de equidad frente a quienes declararon oportunamente.}
\src{\ley{286}{Artículo 70}. Declaración independiente hasta el 30 de junio de 2027. La tarifa grava la base del activo o pasivo, no una renta anual. No legaliza activos de origen ilícito. Activos digitales tienen un régimen separado.}
\end{frame}

\begin{frame}{El artículo 95 abre una vía de alivio a Ecopetrol}
\begin{itemize}\bulletspace
\item La DIAN podrá transar o conciliar tributos de importación de combustibles, por declaraciones hasta diciembre de 2025.
\item Exige disputas superiores a \textbf{2 millones de UVT}, cerca de \textbf{\$104.748 millones} con UVT 2026.
\item Permite usar doctrina más favorable vigente en cada operación. Exige cambio posterior de criterio, proceso sin fallo firme y ausencia de fraude.
\end{itemize}
\take{Ecopetrol y Reficar tienen disputas documentadas. Su beneficio efectivo depende de cada expediente.}
\src{\ley{321}{Artículo 95}. Ecopetrol, comunicados de \href{https://www.ecopetrol.com.co/wps/wcm/connect/e0f637ae-7f70-47c8-bca6-db04c49a68c4/comunicado-dian-esp-original.pdf?MOD=AJPERES}{6 de mayo} y \href{https://www.ecopetrol.com.co/wps/wcm/connect/75535020-ecc2-44b7-afca-c14f3350145d/segundo-comunicado-dian-final-esp-1.pdf?MOD=AJPERES}{26 de junio de 2025}. La norma no las nombra ni limita el beneficio a ellas. El umbral usa UVT al determinar la obligación.}
\end{frame}

\begin{frame}{El ajuste necesita una distribución defendible}
\begin{itemize}\bulletspace
\item El proyecto combina recortes, mayores aportes y cambios tributarios. La necesidad del ajuste sigue vigente.
\item Su efecto depende de ejecutar los recortes, recaudar los ingresos y hacer funcionar la compensación del IVA.
\item Las preferencias de RUGE, la normalización y los tratamientos sectoriales exigen justificar beneficiarios y costos.
\end{itemize}
\take{El Congreso debe corregir preferencias injustificadas y asegurar una devolución del IVA que funcione.}

\end{frame}

\begin{frame}{Anexo. Falta medir la cobertura de las compras}
\begin{itemize}\bulletspace
\item La DIAN dispuso reportes de facturas para \textbf{17,4 millones de personas} en 2025. No informa allí su distribución por ingresos.
\item Una factura a ``consumidor final'' registra la venta, pero no identifica al beneficiario de la devolución.
\item Para evaluar EfectIVA hace falta cruzar compras e IVA identificados con población elegible, territorio y tipo de comercio.
\end{itemize}
\src{DIAN, \href{https://www.dian.gov.co/atencionciudadano/Documents/Informe-de-Gestion-DIAN-2025-30012026.pdf}{Informe de gestión 2025, pp. 14--15}. Personas con reportes disponibles no equivale a compradores frecuentes ni a cobertura de todo su consumo.}
\end{frame}
